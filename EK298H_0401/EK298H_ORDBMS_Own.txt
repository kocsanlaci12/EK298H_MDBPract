1. Feladat– XML adatok kiolvasása XMLTABLE segítségével

a)Az összes autó kilistázása:

SELECT x.autorendszam,
       x.szin,
       x.motor,
       x.karbantartasok,
       x.muszakiallapot
FROM autoszerviz_xml a,
     XMLTABLE('/autoszervizeles/auto'
       PASSING a.adat
       COLUMNS
         autorendszam VARCHAR2(20) PATH '@autorendszam',
         szin VARCHAR2(50) PATH 'szin',
         motor VARCHAR2(100) PATH 'motor',
         karbantartasok NUMBER PATH 'karbantartasok',
         muszakiallapot VARCHAR2(50) PATH 'muszakiallapot'
     ) x;

b)Szűrés: csak a kiválló állapotú autók:

SELECT x.autorendszam,
       x.szin,
       x.motor,
       x.muszakiallapot
FROM autoszerviz_xml a,
     XMLTABLE('/autoszervizeles/auto'
       PASSING a.adat
       COLUMNS
         autorendszam VARCHAR2(20) PATH '@autorendszam',
         szin VARCHAR2(50) PATH 'szin',
         motor VARCHAR2(100) PATH 'motor',
         muszakiallapot VARCHAR2(50) PATH 'muszakiallapot'
     ) x
WHERE x.muszakiallapot = 'Kiváló';

c)Karbantartások kilistázása az autókkal együtt:

SELECT k.a_k_a AS rendszam,
       a.szin,
       a.motor,
       k.munkalatokleirasa,
       k.koltseg,
       k.munkavallaloneve,
       k.datum
FROM autoszerviz_xml x,

     XMLTABLE('/autoszervizeles/karbantartas'
       PASSING x.adat
       COLUMNS
         a_k_a VARCHAR2(20) PATH '@a_k_a',
         munkalatokleirasa VARCHAR2(500) PATH 'munkalatokleirasa',
         koltseg NUMBER PATH 'koltseg',
         munkavallaloneve VARCHAR2(100) PATH 'munkavallaloneve',
         datum DATE PATH 'datum'
     ) k,

     XMLTABLE('/autoszervizeles/auto'
       PASSING x.adat
       COLUMNS
         autorendszam VARCHAR2(20) PATH '@autorendszam',
         szin VARCHAR2(50) PATH 'szin',
         motor VARCHAR2(100) PATH 'motor'
     ) a

WHERE k.a_k_a = a.autorendszam;

2.Feladat: UDT(ObjectType) létrehozása az XML struktúra alapján

a)Cím típus(az XML <lakcim> és <nev> elem alapján):

CREATE OR REPLACE TYPE lakcim_t AS OBJECT (
    iranyitoszam VARCHAR2(10),
    varos VARCHAR2(100),
    utcahazszam VARCHAR2(200)
);

COMMIT;

CREATE OR REPLACE TYPE nev_t AS OBJECT (
    vezeteknev VARCHAR2(100),
    keresztnev VARCHAR2(100)
);

b)Tulajdonos típusbbeágyazott objektummal és tagfüggvénnyel:

CREATE OR REPLACE TYPE tulajdonos_t AS OBJECT (
    tulajid VARCHAR2(10),
    emailcim VARCHAR2(200),
    telefonszam VARCHAR2(20),
    szuletesidatum DATE,
    nev nev_t,
    lakcim lakcim_t,
    
    MEMBER FUNCTION leiras RETURN VARCHAR2
);

c)TYPEBODY tagfüggvény implementálása:

CREATE OR REPLACE TYPE BODY tulajdonos_t AS

    MEMBER FUNCTION leiras RETURN VARCHAR2 IS
    BEGIN
        RETURN SELF.nev.vezeteknev || ' ' || SELF.nev.keresztnev
               || ' – ' || SELF.lakcim.varos
               || ' (' || SELF.emailcim || ')';
    END leiras;

END;

3.Feladat: ObjectTable létrehozása és feltöltése XML adatokból

a)Object Table létrehozása:

CREATE TABLE tulajdonosok OF tulajdonos_t (
    PRIMARY KEY (tulajid)
);

b)Feltöltés XML-ből: INSERT… SELECT + XMLTABLE:

INSERT INTO tulajdonosok
SELECT tulajdonos_t(
    x.tulajid,
    x.emailcim,
    x.telefonszam,
    x.szuletesidatum,
    nev_t(x.vezeteknev, x.keresztnev),
    lakcim_t(x.iranyitoszam, x.varos, x.utcahazszam)
)
FROM autoszerviz_xml a,
     XMLTABLE('/autoszervizeles/tulajdonos'
       PASSING a.adat
       COLUMNS
         tulajid VARCHAR2(10) PATH '@tulajid',
         emailcim VARCHAR2(200) PATH 'emailcim',
         telefonszam VARCHAR2(20) PATH 'telefonszam',
         szuletesidatum DATE PATH 'szuletesidatum',
         vezeteknev VARCHAR2(100) PATH 'nev/vezeteknev',
         keresztnev VARCHAR2(100) PATH 'nev/keresztnev',
         iranyitoszam VARCHAR2(10) PATH 'lakcim/iranyitoszam',
         varos VARCHAR2(100) PATH 'lakcim/varos',
         utcahazszam VARCHAR2(200) PATH 'lakcim/utcahazszam'
     ) x;

c)Ellenőrzés: az Object Table tartalma:

SELECT t.tulajid,
       t.emailcim,
       t.telefonszam,
       t.szuletesidatum,
       t.nev.vezeteknev,
       t.nev.keresztnev,
       t.lakcim.iranyitoszam,
       t.lakcim.varos,
       t.lakcim.utcahazszam
FROM tulajdonosok t;

4.Feladat: Lekérdezések az Object Table-ön

a)Tagfüggvény hívása lekérdezésben:

SELECT t.leiras() AS leiras
FROM tulajdonosok t;

b)Aggregáció–átlagos tulajdonosok száma városonként:

SELECT t.lakcim.varos AS varos,
       COUNT(*) AS db
FROM tulajdonosok t
GROUP BY t.lakcim.varos
ORDER BY db DESC;

5.Feladat: Módosítás és törlés:

a)UDT attribútum módosítása:

UPDATE tulajdonosok t
SET t.emailcim = 'opel.astra@freemail.com'
WHERE t.tulajid = 'T001';

b)30 évesnél fiatalabb tulajdonos törlése:

DELETE FROM tulajdonosok t
WHERE EXTRACT(YEAR FROM SYSDATE) - EXTRACT(YEAR FROM t.szuletesidatum) < 30;

Ellenőrzés:

SELECT COUNT(*) AS maradt
FROM tulajdonosok;

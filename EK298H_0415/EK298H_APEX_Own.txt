1. feladat

Ős típus:
CREATE OR REPLACE TYPE szerviz_elem_t AS OBJECT (
    nev VARCHAR2(100),

    MEMBER FUNCTION bemutatkozas RETURN VARCHAR2,
    STATIC FUNCTION min_kor RETURN NUMBER,
    ORDER MEMBER FUNCTION osszehasonlit(masik szerviz_elem_t) RETURN INTEGER
) NOT FINAL;
/

CREATE OR REPLACE TYPE BODY szerviz_elem_t AS

    MEMBER FUNCTION bemutatkozas RETURN VARCHAR2 IS
    BEGIN
        RETURN 'Nev: ' || SELF.nev;
    END;

    STATIC FUNCTION min_kor RETURN NUMBER IS
    BEGIN
        RETURN 0;
    END;

    ORDER MEMBER FUNCTION osszehasonlit(masik szerviz_elem_t) RETURN INTEGER IS
    BEGIN
        IF SELF.nev < masik.nev THEN RETURN -1;
        ELSIF SELF.nev > masik.nev THEN RETURN 1;
        ELSE RETURN 0;
        END IF;
    END;

END;
/

2. feladat

Autó típus:
CREATE OR REPLACE TYPE auto_t UNDER szerviz_elem_t (
    rendszam VARCHAR2(20),
    szin VARCHAR2(50),
    motor VARCHAR2(50),
    karbantartasok NUMBER,
    muszakiallapot VARCHAR2(50),

    OVERRIDING MEMBER FUNCTION bemutatkozas RETURN VARCHAR2
);
/

CREATE OR REPLACE TYPE BODY auto_t AS
    OVERRIDING MEMBER FUNCTION bemutatkozas RETURN VARCHAR2 IS
    BEGIN
        RETURN 'Auto: ' || SELF.nev || ', rendszam: ' || SELF.rendszam ||
               ', allapot: ' || SELF.muszakiallapot;
    END;
END;
/

3. feladat

Object Table:
CREATE TABLE autoszerviz_tbl OF szerviz_elem_t;

Autók betöltése XML-ből:
INSERT INTO autoszerviz_tbl
SELECT auto_t(
    x.szin,
    x.nev,
    x.rendszam,
    x.motor,
    x.karbantartasok,
    x.muszakiallapot
)
FROM autoszervizeles a,
     XMLTABLE('/autoszervizeles/auto'
     PASSING a.adat
     COLUMNS
        rendszam VARCHAR2(20) PATH '@autorendszam',
        szin VARCHAR2(50) PATH 'szin',
        motor VARCHAR2(50) PATH 'motor',
        karbantartasok NUMBER PATH 'karbantartasok',
        muszakiallapot VARCHAR2(50) PATH 'muszakiallapot',
        nev VARCHAR2(100) PATH 'motor'
     ) x;

Szervizek betöltése:
INSERT INTO autoszerviz_tbl

4.feladat:

SELECT szerviz_elem_t.min_kor() FROM dual;

Polimorf hívás:
SELECT s.nev, s.bemutatkozas()
FROM autoszerviz_tbl s
ORDER BY VALUE(s);

Szűrés típus szerint:
SELECT s.nev
FROM autoszerviz_tbl s
WHERE VALUE(s) IS OF (auto_t);

Típus konverzió:
SELECT s.nev,
       TREAT(VALUE(s) AS auto_t).muszakiallapot
FROM autoszerviz_tbl s
WHERE VALUE(s) IS OF (auto_t);

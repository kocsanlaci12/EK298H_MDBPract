Lookup – autók és karbantartások összekapcsolása

db.karbantartas.aggregate([
{
$lookup: {
from: "auto",
localField: "_a_k_a",
foreignField: "_autorendszam",
as: "auto_adatok"
}
},
{ $unwind: "$auto_adatok" },
{ $match: { "auto_adatok.muszakiallapot": "Kiváló" } },
{
$project: {
_id: 0,
munkavallaloneve: 1,
datum: 1,
auto: "$auto_adatok._autorendszam",
allapot: "$auto_adatok.muszakiallapot"
}
}
])
